cmake_minimum_required(VERSION 3.16)
project(clockwork_relay CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release)
endif()

add_executable(clockwork-relay src/relay.cpp src/relay_main.cpp)
if(WIN32)
    target_link_libraries(clockwork-relay PRIVATE ws2_32)
endif()
if(MSVC)
    target_compile_options(clockwork-relay PRIVATE /W4)
else()
    target_compile_options(clockwork-relay PRIVATE -Wall -Wextra)
endif()
install(TARGETS clockwork-relay)
